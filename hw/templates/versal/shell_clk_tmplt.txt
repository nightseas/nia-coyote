# Shell clock
# On UltraScale+ devices, the XDMA produces a fixed clock at 250 MHz
# On Versal devices, the static layer frequency (xclk) can be set during CMake config, through the parameter SCLK_F
create_clock -period {{ cnfg.sclk_p }} [get_ports xclk];

# There should be no clock buffers when crossing PR boundaries (Vivado 73360)
set_property CLOCK_BUFFER_TYPE NONE [get_ports xclk]
